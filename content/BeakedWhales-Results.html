#Import beaked whale detections
detections_summary<-readRDS(here('_data','BeakedWhales','Detections_summary.rds'))
# Acoustic study of all beaked whale detections
data <-readRDS(here('_data','BeakedWhales','CombinedStudy_EventswRecordingsGPS.rds'))
# Click detection data
detections<-readRDS(here('_data','BeakedWhales','Detections.rds'))

#Import deployment info
deployment_info<-readRDS(here('_data','BeakedWhales','Deployment_info.rds'))

#Calculate recording effort 
recordings <- data@files$recordings %>%
  mutate(DriftName = stringr::str_extract(file, "CalCurCEAS_\\d{3}"))

recordings_unique <- recordings %>%
  distinct(file, .keep_all = TRUE)

effort_deployment <- recordings_unique %>%
  group_by(DriftName) %>%
  summarise(
    recording_hours = sum(length, na.rm = TRUE) / 3600,
    n_files = n(),
    .groups = "drop")

## Total recording effort
effort_total <- effort_deployment %>%
  summarise(
    n_drifts = n(),
    recording_hours = sum(recording_hours),
    recording_days = recording_hours / 24)

#Encounter rates
encounters <-detections %>%
  group_by(eventId) %>%
  summarise(
    DriftName = first(.data$DriftName),
    species = first(.data$species),
    encounter_start = min(.data$UTC, na.rm = TRUE),
    encounter_end = max(.data$UTC, na.rm = TRUE),
    encounter_duration_min = as.numeric(
      difftime(encounter_end, encounter_start, units = "mins")
    ),
    n_clicks = n(),
    .groups = "drop")

encounter_rates <- encounters %>%
  count(DriftName, name = "n_encounters") %>%
  right_join(effort_deployment, by = "DriftName") %>%
  mutate(
    n_encounters = tidyr::replace_na(n_encounters, 0),
    encounters_per_24h = n_encounters / recording_hours * 24)

overall_encounter_rate <- nrow(encounters) /
  sum(effort_deployment$recording_hours) * 24

##Sort drifts by encounter rates 
#arrange(encounter_rates, desc(encounters_per_24h))

#Identify mixed species groups
# 1. Group encounters that occur within 20 min of one another
encounters_grouped <- encounters %>%
  arrange(DriftName, encounter_start) %>%
  group_by(DriftName) %>%
  mutate(
    gap_min = as.numeric(
      difftime(
        encounter_start,
        lag(encounter_end),
        units = "mins"
      )
    ),
    new_encounter_group = is.na(gap_min) | gap_min > 20,
    encounter_group = cumsum(new_encounter_group)
  ) %>%
  ungroup()


# 2. Summarize each encounter group
mixed_encounters <- encounters_grouped %>%
  group_by(DriftName, encounter_group) %>%
  summarise(
    group_start = min(encounter_start),
    group_end = max(encounter_end),
    n_encounters = n(),
    n_species = n_distinct(species),
    species_present = paste(
      sort(unique(species)),
      collapse = ", "
    ),
    encounter_ids = paste(eventId, collapse = ", "),
    .groups = "drop"
  ) %>%
  mutate(
    mixed_species = n_species > 1)


# 3. Summarize mixed-species encounters by drift
mixed_by_drift <- mixed_encounters %>%
  group_by(DriftName) %>%
  summarise(
    n_encounter_groups = n(),
    n_mixed_encounters = sum(mixed_species),
    percent_mixed = 100 * n_mixed_encounters / n_encounter_groups,
    .groups = "drop"
  )

mixed_summary <- mixed_by_drift %>%
  summarise(
    total_encounters = sum(n_encounter_groups),
    total_mixed = sum(n_mixed_encounters),
    percent_mixed = 100 * total_mixed / total_encounters,
    drifts_with_mixed = sum(n_mixed_encounters > 0),
    max_mixed_per_drift = max(n_mixed_encounters)
  )

Overall beaked whale presence

There were {r}length(unique(detections_summmary$eventId total beaked whale events detected on 22 separate drifts, corresponding to an overall encounter rate of {r}overall_encounter_rate encounters per 24 hours of recording. Encounter rates varied among individual drifts, ranging from 0-6.9 encounters per 24 hours (median = 1.9, IQR = [0.9-4.1]. The highest encounter rates were observed during drifts 10 (6.9 encounters per 24 h), drift 22 (6.7 encounters per 24 h), and drift 9 (6.2 encounters per 24 h), while drifts 12, 13, and 14 contained zero beaked whale acoustic detections.

Five distinct beaked whale species were acoustically detected, including goose-beaked, Hubb’s, Stejneger’s, Baird’s, and gingko-toothed beaked whales [@tbl_species_detections]. Encounters with multiple beaked whales were relatively uncommon, comprising {r}mixed_summary$total_mixed of {r}mixed_summary$total_encounters encounters ({r}mixed_summary$percent_mixed %. Mixed-species of beaked whale encounters occurred on {r}mixed_summary$drifts_with_mixed drifts, with 0-{r}mixed_summary$max_mixed_per_drift mixed species encounters per drift. The greatest numbers of mixed species were observed on drifts 9 (7 encounters) and drift 10 (7 encounters).

Analyst comments noted dolphins at the surface during {r}dolphin_summary$n_encounters beaked whale encounters across {r}dolphin_summary$n_driftsdrifts. Because these observations were recorded opportunistically during acoustic review rather than through a standardized protocol, they are reported as incidental observations rather than as a rate or proportion of encounters.

species_summary_pub <- detections_summary %>%
  group_by(species) %>%
  summarise(
    `Acoustic events` = n(),
    `Total duration (min)` = round(sum(Duration_min, na.rm = TRUE), 0),
    `Median event duration (min)` = round(median(Duration_min, na.rm = TRUE), 0),
    `Total clicks` = sum(n_clicks, na.rm = TRUE),
    .groups = "drop"
  ) %>%
  mutate(
    Species = recode(
      species,
      "BBWH"  = "*Berardius bairdii*",
      "GOBW"  = "*Ziphius cavirostris*",
      "HUBW"  = "*Mesoplodon carlhubbsi*",
      "GTBW"  = "*Mesoplodon ginkgodens*",
      "STBWH" = "*Mesoplodon stejnegeri*", 
      "UNME"  = "Unidentified beaked whale"
    )
  ) %>%
  select(
    Species,
    `Acoustic events`,
    `Total duration (min)`,
    `Median event duration (min)`,
    `Total clicks`
  ) %>%
  arrange(Species)

detections_summary <- detections_summary %>%
  mutate(date = as.Date(UTC))

sun_times <- getSunlightTimes(
  data = detections_summary %>%
    transmute(
      date = date,
      lat = Latitude,
      lon = Longitude),
  keep = c("dawn", "sunrise", "sunset", "dusk"),
  tz = "UTC")

#bind to detections summary
detections_summary_diel <- detections_summary %>%
  bind_cols(
    sun_times %>%
      select(dawn, sunrise, sunset, dusk))

#assign diel category to each event
detections_summary_diel <- detections_summary_diel %>%
  mutate(
    diel_period = case_when(
      UTC >= sunrise & UTC < sunset ~ "day",
      UTC >= dawn & UTC < sunrise ~ "dawn",
      UTC >= sunset & UTC < dusk ~ "dusk",
      TRUE ~ "night"))

detections_summary_diel <- detections_summary_diel %>%
  mutate(
    diel_period = factor(
      diel_period,
      levels = c("dawn", "day", "dusk", "night")))


#Calculate total detected minutes per diel period
detected_minutes <- detections_summary_diel %>%
  group_by(species, diel_period) %>%
  summarise(
    detected_min = sum(Duration_min, na.rm = TRUE),
    n_events = n(),
